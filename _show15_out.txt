function buildPhotosGridHtml(items, imageSrcs) {
  let rows = "";
  for (let i = 0; i < items.length; i += 2) {
    const cell1 = buildPhotoCellHtml(i + 1, items[i], imageSrcs[i]);
    const cell2 = (i + 1 < items.length)
      ? buildPhotoCellHtml(i + 2, items[i + 1], imageSrcs[i + 1])
      : "<td style=\"width:50%;\"></td>";
    rows += "<tr style=\"page-break-inside:avoid;\">" + cell1 + cell2 + "</tr>";
  }
  return "<table style=\"width:100%; border-collapse:collapse;\">" + rows + "</table>";
}
