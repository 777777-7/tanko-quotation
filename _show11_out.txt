function escapeHtml(str) {
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

// Product names are stored as "\n"-joined feature lines (one feature per
// line). This renders each line separately instead of mixing them together.
function nl2br(str) {
  return escapeHtml(str).replace(/\n/g, "<br>");
}

function formatFeatureLine(f) {
  const trimmed = String(f).trim();
  const upper = trimmed.toUpperCase();
  if (COLOUR_WORDS.has(upper)) return "Colour: " + trimmed;
  if (MATERIAL_WORDS.has(upper)) return "Material: " + (upper === "S STEEL" ? "Stainless Steel" : trimmed);
  return f;
}

// Re-orders a list of feature lines so the "Size:" line always comes first,
// whatever order it happens to sit in inside PRODUCT_DATA.
function orderFeatureLines(lines) {
  const i = lines.findIndex((l) => /^Size(?: \(approx\))?\s*:/i.test(String(l).trim()));
  if (i <= 0) return lines;
  const out = lines.slice();
  out.unshift(out.splice(i, 1)[0]);
  return out;
}
